

# Command to create the keys

openssl req -x509 -nodes -days 365 -newkey rsa:2048 \
  -keyout tls.key -out tls.crt \
  -subj "/CN=mytestdemo.local/O=Development"

# Push those local files into Kubernetes

- kubectl create secret tls my-tls-secret --key=tls.key --cert=tls.crt

# remove the from the local

- rm tls.key tls.crt

# Kubernetes successfully received and secured your keys.

- kubectl get secrets

# Without enabling this addon, your ingress.yaml file is just a piece of text sitting in your cluster

- minikube addons enable ingress

# minikube tunnel is meant to keep loading. It functions as a persistent background routing process. A blank, blinking terminal cursor means the network tunnel is open and active.
It grabs Ports 80 and 443 on your actual laptop interface (127.0.0.1)


- "minikube tunnel" - Meant for type: LoadBalancer services
- "minikube service <name>" - Meant for type: NodePort services

# to change the controller's listening ports permanently

kubectl patch deployment ingress-nginx-controller -n ingress-nginx --type='json' -p='[{"op": "replace", "path": "/spec/template/spec/containers/0/ports/1/hostPort", "value": 8443}]'

# Ensure your port forward command is still running in its separate window

- kubectl port-forward deployment/ingress-nginx-controller -n ingress-nginx 8443:443
